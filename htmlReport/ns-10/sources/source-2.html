 fun convertToFoodInfoList(cartItems: List<CartItem>): List<FoodInfo> {
     return cartItems.map { cartItem ->
         FoodInfo(
             name = cartItem.foodItem.name,
             quantity = cartItem.quantity,
             type = cartItem.foodItem.foodType
         )
     }
